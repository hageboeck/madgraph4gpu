get_filename_component(PROCESS ${CMAKE_CURRENT_SOURCE_DIR} NAME)
set(libname_cxx mg5amc_cxx_${PROCESS})
set(libname_f mg5amc_f_${PROCESS})

# List files that can be compiled either as C++ or Cuda
set(CXX_AND_CUDA_FILES
  BridgeKernels.cc
  CommonRandomNumberKernel.cc
  counters.cc
  CPPProcess.cc
  CrossSectionKernels.cc
  MatrixElementKernels.cc
  RamboSamplingKernels.cc
  fbridge.cc
  fsampler.cc
)

######################################################
# Common library for this process:
######################################################
add_library(${libname_cxx} OBJECT ${CXX_AND_CUDA_FILES})
target_include_directories(${libname_cxx} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(${libname_cxx} PUBLIC mg5amc_common)

if(TARGET CUDA::curand)
  target_sources(${libname_cxx} PRIVATE CurandRandomNumberKernel.cc)
  target_link_libraries(${libname_cxx} PUBLIC CUDA::curand)
  list(APPEND CXX_AND_CUDA_FILES CurandRandomNumberKernel.cc)
else()
  target_compile_definitions(${libname_cxx} PUBLIC MGONGPU_HAS_NO_CURAND)
endif()

if(HIP_FOUND)
  message(WARNING "Trying to configure HIP random numbers (untested):")
  target_sources(${libname_cxx} PRIVATE HiprandRandomNumberKernel.cc)
  target_link_libraries(${libname_cxx} PUBLIC hip::hiprand)
else()
  target_compile_definitions(${libname_cxx} PUBLIC MGONGPU_HAS_NO_HIPRAND)
endif()


######################################################
# Check executable:
######################################################
set(execname ${PROCESS}_check.exe)
add_executable(${execname} check_sa.cc)
target_link_libraries(${execname} PUBLIC ${libname_cxx})
set_property(TARGET ${execname} PROPERTY OUTPUT_NAME check.exe)

# Change C++ compilation into CUDA compilation:
if(DEFINED MG_ENABLE_LANGUAGE)
  set_source_files_properties(${CXX_AND_CUDA_FILES} PROPERTIES LANGUAGE ${MG_ENABLE_LANGUAGE})
  set_source_files_properties(check_sa.cc PROPERTIES LANGUAGE ${MG_ENABLE_LANGUAGE})
endif()

######################################################
# Madevent:
######################################################

add_library(${libname_f} OBJECT
  addmothers.f
  # auto_dsig1.f
  auto_dsig.f
  cluster.f
  cuts.f
  dummy_fct.f
  genps.f
  get_color.f
  idenparts.f
  initcluster.f
  myamp.f
  unwgt.f
  reweight.f
  setcuts.f
  setscales.f
  ompnumthreads.cc
  matrix1.f
#  symmetry.f
)
target_include_directories(${libname_f} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(${libname_f} PUBLIC Source)


# Madevent
set(execname ${PROCESS}_madevent_fortran.exe)
add_executable(${execname} auto_dsig1.f driver.f)
target_link_libraries(${execname} PUBLIC ${libname_f} ${libname_cxx})
set_property(TARGET ${execname} PROPERTY OUTPUT_NAME madevent_fortran)

set(execname ${PROCESS}_madevent_cxx.exe)
add_executable(${execname} auto_dsig1.f driver.f)
target_link_libraries(${execname} PUBLIC ${libname_f} ${libname_cxx})
set_property(TARGET ${execname} PROPERTY OUTPUT_NAME madevent_cpp)
target_compile_definitions(${execname} PUBLIC MG5AMC_MEEXPORTER_CUDACPP)
